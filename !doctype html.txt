<script>
const LABELS = ["Built-up","Agriculture","Vegetation","Water","Other","Unsure"];
const KEYS = {"1":0,"2":1,"3":2,"4":3,"5":4,"6":5};
let rows = [], header = [], idx = 0, storeKey = "";

const map = L.map("map", { zoomControl:true, maxZoom:19 });
L.tileLayer("https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}",
  { maxZoom:19, attribution:"Imagery &copy; Esri, Maxar, Earthstar Geographics" }).addTo(map);
let box = null, outer = null;

function parseCSV(text) {
  const lines = text.replace(/^\uFEFF/, "").split(/\r?\n/).filter(l => l.trim() !== "");
  const hdr = lines[0].split(",").map(s => s.trim());
  const out = lines.slice(1).map(l => {
    const c = l.split(",");
    const o = {}; hdr.forEach((h,i) => o[h] = (c[i] ?? "").trim());
    return o;
  });
  return { hdr, out };
}

function save() { try { localStorage.setItem(storeKey, JSON.stringify({idx, labels: rows.map(r => r.true_class)})); } catch(e){} }

function load(text) {
  const p = parseCSV(text);
  for (const need of ["id","lon","lat","true_class"])
    if (!p.hdr.includes(need)) { alert("Missing column: " + need); return; }
  header = p.hdr; rows = p.out;
  storeKey = "labeller:" + rows.length + ":" + rows[0].id + ":" + rows[0].lon + ":" + rows[rows.length-1].lon;
  try {
    const s = JSON.parse(localStorage.getItem(storeKey) || "null");
    if (s && s.labels && s.labels.length === rows.length) {
      s.labels.forEach((v,i) => { if (v) rows[i].true_class = v; });
      idx = Math.min(s.idx || 0, rows.length - 1);
    } else {
      idx = Math.max(0, rows.findIndex(r => !r.true_class));
    }
  } catch(e) { idx = 0; }
  document.getElementById("empty").style.display = "none";
  show();
}

function show() {
  const r = rows[idx]; if (!r) return;
  const lat = parseFloat(r.lat), lon = parseFloat(r.lon);
  map.setView([lat, lon], 18);
  const dLat = 10 / 111320, dLon = 10 / (111320 * Math.cos(lat * Math.PI / 180));
  if (box) map.removeLayer(box); if (outer) map.removeLayer(outer);
  box = L.rectangle([[lat-dLat, lon-dLon],[lat+dLat, lon+dLon]],
        { color:"#ff9800", weight:2, fill:false }).addTo(map);
  outer = L.rectangle([[lat-5*dLat, lon-5*dLon],[lat+5*dLat, lon+5*dLon]],
        { color:"#ffffff", weight:1, dashArray:"4", fill:false, opacity:.5 }).addTo(map);
  const done = rows.filter(x => x.true_class).length;
  document.getElementById("pt").innerHTML = `<b>Point ${r.id}</b> (${idx+1} of ${rows.length})<br>${r.lat}, ${r.lon}`;
  document.getElementById("progress").textContent = `${done} / ${rows.length} labelled`;
  document.getElementById("fill").style.width = (100 * done / rows.length) + "%";
  document.getElementById("current").innerHTML = r.true_class ? `Current label: <b>${r.true_class}</b>` : "Not labelled yet";
  document.querySelectorAll(".lab").forEach(b => b.classList.toggle("on", b.dataset.l === r.true_class));
  document.getElementById("links").innerHTML =
    `<a style="color:#4c9aff" target="_blank" href="https://earth.google.com/web/search/${r.lat},${r.lon}">Open in Google Earth</a>`;
  save();
}

function setLabel(l) {
  if (!rows.length) return;
  rows[idx].true_class = l;
  if (idx < rows.length - 1) idx++;
  else alert("That was the last point. Click Export CSV to save.");
  show();
}
function go(d) { if (!rows.length) return; idx = Math.min(rows.length-1, Math.max(0, idx + d)); show(); }

function exportCSV() {
  if (!rows.length) return;
  const lines = [header.join(",")].concat(rows.map(r => header.map(h => r[h] ?? "").join(",")));
  const blob = new Blob([lines.join("\r\n") + "\r\n"], { type:"text/csv;charset=utf-8" });
  const a = document.createElement("a");
  a.href = URL.createObjectURL(blob); a.download = "validation_points_swir.csv"; a.click();
}

document.getElementById("file").addEventListener("change", e => {
  const f = e.target.files[0]; if (!f) return;
  const rd = new FileReader(); rd.onload = () => load(rd.result); rd.readAsText(f);
});
document.querySelectorAll(".lab").forEach(b => b.addEventListener("click", () => setLabel(b.dataset.l)));
document.getElementById("prev").onclick = () => go(-1);
document.getElementById("skip").onclick = () => go(1);
document.getElementById("export").onclick = exportCSV;
document.addEventListener("keydown", e => {
  if (e.key in KEYS) setLabel(LABELS[KEYS[e.key]]);
  else if (e.key === "Backspace" || e.key === "ArrowLeft") { e.preventDefault(); go(-1); }
  else if (e.key === "ArrowRight") go(1);
  else if (e.key.toLowerCase() === "e") exportCSV();
});
map.setView([28.5, 77.5], 8);
</script>
